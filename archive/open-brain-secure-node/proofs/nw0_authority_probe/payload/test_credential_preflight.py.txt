from __future__ import annotations

import asyncio
import threading
import unittest

import authority_gate as authority
import test_authority_gate as test_support

FakeAccess = test_support.FakeAccess


class CredentialPreflightTests(unittest.TestCase):
    def setUp(self) -> None:
        self.case = test_support.AuthorityGateTests()
        self.case.setUp()

    def tearDown(self) -> None:
        self.case.tearDown()

    def assert_unused(self, reservation: str) -> None:
        snapshot = self.case.gate.snapshot()
        attempt = next(
            item for item in snapshot["attempts"] if item["reservation_id"] == reservation
        )
        self.assertEqual(attempt["status"], "cancelled")
        self.assertEqual(snapshot["charged_attempts"], 0)
        self.assertNotIn("dispatch_intent", [item["kind"] for item in snapshot["events"]])

    def test_missing_credential_cancels_proven_unused_reservation(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        access = FakeAccess()

        def missing() -> str:
            access.resolve_calls += 1
            return ""

        access.resolve = missing
        with self.assertRaisesRegex(authority.Rejected, "credential unavailable"):
            self.case.release(capability, reservation, access)
        self.assertEqual(access.calls, (1, 0, 0))
        self.assert_unused(reservation)

    def test_resolver_exception_cancels_without_provider_construction(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        access = FakeAccess()

        def fail() -> str:
            access.resolve_calls += 1
            raise RuntimeError("synthetic credential store failure")

        access.resolve = fail
        with self.assertRaisesRegex(RuntimeError, "credential store"):
            self.case.release(capability, reservation, access)
        self.assertEqual(access.calls, (1, 0, 0))
        self.assert_unused(reservation)

    def test_revoke_during_preflight_rechecks_before_intent(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        access = FakeAccess(
            on_resolve=lambda: self.case.gate.revoke_consent(
                capability, provider="openai_api"
            )
        )
        with self.assertRaises(authority.Rejected):
            self.case.release(capability, reservation, access)
        self.assertEqual(access.calls, (1, 0, 0))
        self.assert_unused(reservation)

    def test_revision_during_preflight_rechecks_before_intent(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        access = FakeAccess(
            on_resolve=lambda: self.case.gate.observe_revision(
                self.case.bindings[0].note_id, "new-revision"
            )
        )
        with self.assertRaises(authority.Rejected):
            self.case.release(capability, reservation, access)
        self.assertEqual(access.calls, (1, 0, 0))
        self.assert_unused(reservation)

    def test_authority_denial_precedes_credential_resolution(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        self.case.gate.revoke_consent(capability, provider="openai_api")
        access = FakeAccess()
        with self.assertRaises(authority.Rejected):
            self.case.release(capability, reservation, access)
        self.assertEqual(access.calls, (0, 0, 0))
        self.assert_unused(reservation)

    def test_duplicate_release_race_resolves_and_constructs_once(self) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability)
        resolving = threading.Event()
        proceed = threading.Event()

        def hold() -> None:
            resolving.set()
            self.assertTrue(proceed.wait(2))

        access = FakeAccess(on_resolve=hold)
        results: list[str] = []
        errors: list[str] = []

        def first() -> None:
            results.append(self.case.release(capability, reservation, access))

        thread = threading.Thread(target=first)
        thread.start()
        self.assertTrue(resolving.wait(2))
        try:
            self.case.release(capability, reservation, access)
        except authority.Rejected as error:
            errors.append(str(error))
        proceed.set()
        thread.join(2)
        self.assertFalse(thread.is_alive())
        self.assertEqual(results, ['{"edges":[]}'])
        self.assertEqual(errors, ["credential preflight active"])
        self.assertEqual(access.calls, (1, 1, 1))


class AsyncCredentialPreflightTests(unittest.IsolatedAsyncioTestCase):
    def setUp(self) -> None:
        self.case = test_support.AuthorityGateTests()
        self.case.setUp()

    def tearDown(self) -> None:
        self.case.tearDown()

    async def release(self, resolver) -> None:
        capability = self.case.authorize()
        reservation = self.case.reserve(capability, timeout_seconds=1)
        constructions = []
        with self.assertRaises((TimeoutError, asyncio.CancelledError)):
            await self.case.gate.release_async(
                capability,
                reservation_id=reservation,
                provider="openai_api",
                adapter_identity="openai_api:synthetic-v1",
                operation="semantic_graph",
                selected_sources=self.case.selected(),
                resolve_credential=resolver,
                construct_provider=lambda *_: constructions.append(True),
            )
        self.assertEqual(constructions, [])
        snapshot = self.case.gate.snapshot()
        self.assertEqual(snapshot["attempts"][0]["status"], "cancelled")
        self.assertEqual(snapshot["charged_attempts"], 0)
        self.assertEqual(self.case.gate._credential_preflights, set())

    async def test_async_resolver_timeout_cleans_inline(self) -> None:
        cleaned = asyncio.Event()

        async def resolver() -> str:
            try:
                await asyncio.Event().wait()
            finally:
                cleaned.set()

        await self.release(resolver)
        self.assertTrue(cleaned.is_set())

    async def test_owner_cancel_cleans_async_resolver_inline(self) -> None:
        started = asyncio.Event()
        cleaned = asyncio.Event()

        async def resolver() -> str:
            started.set()
            try:
                await asyncio.Event().wait()
            finally:
                cleaned.set()

        owner = asyncio.create_task(self.release(resolver))
        await started.wait()
        owner.cancel()
        await owner
        self.assertTrue(cleaned.is_set())


if __name__ == "__main__":
    unittest.main(verbosity=2)
