"""Synthetic loopback-only DNS and TLS fixtures for the E43 proof."""

from __future__ import annotations

import asyncio
import contextlib
import datetime
import ipaddress
import socket
import ssl
import struct
import tempfile
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from cryptography import x509
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.x509.oid import NameOID

LOGICAL_HOST = "transport.test"


def _question_end(packet: bytes) -> int:
    index = 12
    while index < len(packet):
        size = packet[index]
        index += 1
        if size == 0:
            if index + 4 > len(packet):
                raise ValueError("short question")
            return index + 4
        if size > 63 or index + size > len(packet):
            raise ValueError("invalid label")
        index += size
    raise ValueError("unterminated question")


def _dns_response(packet: bytes, mode: str) -> bytes | None:
    if mode == "drop":
        return None
    if mode == "malformed":
        return packet[:2] + b"\x81"
    end = _question_end(packet)
    question = packet[12:end]
    answer = b"\xc0\x0c" + struct.pack("!HHIH", 1, 1, 60, 4) + socket.inet_aton("127.0.0.1")
    return packet[:2] + struct.pack("!HHHHH", 0x8180, 1, 1, 0, 0) + question + answer


class _DnsProtocol(asyncio.DatagramProtocol):
    def __init__(self, mode: str) -> None:
        self.mode = mode
        self.queries = 0
        self.errors: list[str] = []
        self.transport: asyncio.DatagramTransport | None = None

    def connection_made(self, transport: asyncio.BaseTransport) -> None:
        if not isinstance(transport, asyncio.DatagramTransport):
            raise TypeError("datagram transport required")
        self.transport = transport

    def datagram_received(self, data: bytes, addr: tuple[str, int]) -> None:
        self.queries += 1
        try:
            response = _dns_response(data, self.mode)
            if response is not None and self.transport is not None:
                self.transport.sendto(response, addr)
        except Exception as error:
            self.errors.append(type(error).__name__)

    def error_received(self, exc: Exception) -> None:
        self.errors.append(type(exc).__name__)


@dataclass(slots=True)
class DnsFixture:
    mode: str
    transport: asyncio.DatagramTransport
    protocol: _DnsProtocol
    port: int

    @classmethod
    async def start(cls, mode: str) -> DnsFixture:
        if mode not in {"success", "drop", "malformed"}:
            raise ValueError("DNS fixture mode")
        loop = asyncio.get_running_loop()
        transport, protocol = await loop.create_datagram_endpoint(
            lambda: _DnsProtocol(mode), local_addr=("127.0.0.1", 0)
        )
        address = transport.get_extra_info("sockname")
        if not isinstance(address, tuple) or address[0] != "127.0.0.1":
            transport.close()
            raise RuntimeError("DNS fixture escaped loopback")
        return cls(mode, transport, protocol, int(address[1]))

    async def close(self) -> None:
        self.transport.close()
        await asyncio.sleep(0)
        if self.protocol.errors:
            raise RuntimeError("DNS fixture error")


@dataclass(slots=True)
class TlsMaterial:
    temporary: tempfile.TemporaryDirectory[str]
    ca_pem: bytes
    certificate: Path
    private_key: Path

    @classmethod
    def create(cls) -> TlsMaterial:
        temporary = tempfile.TemporaryDirectory(prefix="e43-synthetic-tls-")
        root = Path(temporary.name)
        now = datetime.datetime.now(datetime.UTC)
        ca_key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        ca_name = x509.Name([x509.NameAttribute(NameOID.COMMON_NAME, "E43 synthetic CA")])
        ca = (
            x509.CertificateBuilder()
            .subject_name(ca_name)
            .issuer_name(ca_name)
            .public_key(ca_key.public_key())
            .serial_number(x509.random_serial_number())
            .not_valid_before(now - datetime.timedelta(minutes=1))
            .not_valid_after(now + datetime.timedelta(hours=1))
            .add_extension(x509.BasicConstraints(ca=True, path_length=0), critical=True)
            .add_extension(
                x509.KeyUsage(
                    digital_signature=False,
                    content_commitment=False,
                    key_encipherment=False,
                    data_encipherment=False,
                    key_agreement=False,
                    key_cert_sign=True,
                    crl_sign=True,
                    encipher_only=False,
                    decipher_only=False,
                ),
                critical=True,
            )
            .sign(ca_key, hashes.SHA256())
        )
        server_key = rsa.generate_private_key(public_exponent=65537, key_size=2048)
        server_name = x509.Name(
            [x509.NameAttribute(NameOID.COMMON_NAME, "E43 loopback server")]
        )
        server = (
            x509.CertificateBuilder()
            .subject_name(server_name)
            .issuer_name(ca_name)
            .public_key(server_key.public_key())
            .serial_number(x509.random_serial_number())
            .not_valid_before(now - datetime.timedelta(minutes=1))
            .not_valid_after(now + datetime.timedelta(hours=1))
            .add_extension(x509.BasicConstraints(ca=False, path_length=None), critical=True)
            .add_extension(
                x509.SubjectAlternativeName(
                    [x509.DNSName(LOGICAL_HOST), x509.IPAddress(ipaddress.ip_address("127.0.0.1"))]
                ),
                critical=False,
            )
            .sign(ca_key, hashes.SHA256())
        )
        certificate = root / "server.pem"
        private_key = root / "server.key"
        certificate.write_bytes(server.public_bytes(serialization.Encoding.PEM))
        private_key.write_bytes(
            server_key.private_bytes(
                serialization.Encoding.PEM,
                serialization.PrivateFormat.PKCS8,
                serialization.NoEncryption(),
            )
        )
        certificate.chmod(0o600)
        private_key.chmod(0o600)
        return cls(
            temporary,
            ca.public_bytes(serialization.Encoding.PEM),
            certificate,
            private_key,
        )

    def client_context(self) -> ssl.SSLContext:
        context = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
        context.check_hostname = True
        context.verify_mode = ssl.CERT_REQUIRED
        context.minimum_version = ssl.TLSVersion.TLSv1_2
        context.load_verify_locations(cadata=self.ca_pem.decode("ascii"))
        return context

    def server_context(self, sni: list[str | None]) -> ssl.SSLContext:
        context = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        context.minimum_version = ssl.TLSVersion.TLSv1_2
        context.load_cert_chain(self.certificate, self.private_key)
        context.set_servername_callback(lambda _sock, name, _context: sni.append(name))
        return context

    def close(self) -> None:
        self.temporary.cleanup()


@dataclass(slots=True)
class TlsFixture:
    mode: str
    server: asyncio.Server
    material: TlsMaterial
    port: int
    sni: list[str | None] = field(default_factory=list)
    bodies: list[bytes] = field(default_factory=list)
    errors: list[str] = field(default_factory=list)
    tasks: set[asyncio.Task[Any]] = field(default_factory=set)
    blocker: asyncio.Event = field(default_factory=asyncio.Event)

    @classmethod
    async def start(cls, mode: str) -> TlsFixture:
        if mode not in {"success", "stall", "oversize"}:
            raise ValueError("TLS fixture mode")
        material = TlsMaterial.create()
        holder: dict[str, TlsFixture] = {}

        async def handle(reader: asyncio.StreamReader, writer: asyncio.StreamWriter) -> None:
            fixture = holder["fixture"]
            task = asyncio.current_task()
            if task is not None:
                fixture.tasks.add(task)
            try:
                if mode == "stall":
                    await fixture.blocker.wait()
                    return
                header = await reader.readuntil(b"\r\n\r\n")
                lengths = [
                    line.split(b":", 1)[1].strip()
                    for line in header.split(b"\r\n")[1:]
                    if line.lower().startswith(b"content-length:")
                ]
                if len(lengths) != 1 or not lengths[0].isdigit():
                    raise ValueError("request length")
                body = await reader.readexactly(int(lengths[0]))
                fixture.bodies.append(body)
                response = b"x" * (20 * 1024) if mode == "oversize" else b'{"ok":true}'
                writer.write(
                    b"HTTP/1.1 200 OK\r\nConnection: close\r\nContent-Length: "
                    + str(len(response)).encode("ascii")
                    + b"\r\n\r\n"
                    + response
                )
                await writer.drain()
            except Exception as error:
                fixture.errors.append(type(error).__name__)
            finally:
                writer.close()
                with contextlib.suppress(Exception):
                    await writer.wait_closed()
                if task is not None:
                    fixture.tasks.discard(task)

        sni: list[str | None] = []
        if mode == "stall":
            server = await asyncio.start_server(handle, "127.0.0.1", 0)
        else:
            server = await asyncio.start_server(
                handle,
                "127.0.0.1",
                0,
                ssl=material.server_context(sni),
                ssl_handshake_timeout=1.0,
            )
        address = server.sockets[0].getsockname()
        fixture = cls(mode, server, material, int(address[1]), sni=sni)
        holder["fixture"] = fixture
        return fixture

    async def close(self) -> None:
        self.blocker.set()
        self.server.close()
        await self.server.wait_closed()
        if self.tasks:
            done, pending = await asyncio.wait(self.tasks, timeout=1.0)
            if pending:
                for task in pending:
                    task.cancel()
                await asyncio.gather(*pending, return_exceptions=True)
            for task in done:
                if not task.cancelled():
                    task.exception()
        self.material.close()
