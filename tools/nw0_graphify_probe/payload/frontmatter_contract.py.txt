"""Private Open Brain preflight prototype; no shipping Engine API is defined here."""
import datetime
import json
import math
import yaml
from graphify.extractors.markdown import _split_frontmatter

class MetadataRejected(ValueError):
    pass

class ContractLoader(yaml.SafeLoader):
    def construct_mapping(self, node, deep=False):
        if not isinstance(node, yaml.MappingNode):
            raise MetadataRejected('mapping_required')
        result = {}
        for key_node, value_node in node.value:
            if not isinstance(key_node, yaml.ScalarNode) or key_node.tag != 'tag:yaml.org,2002:str':
                raise MetadataRejected('string_keys_required')
            key = key_node.value
            if key in result:
                raise MetadataRejected('duplicate_key')
            if node is self.root_node and key == 'aliases':
                if not isinstance(value_node, yaml.SequenceNode):
                    raise MetadataRejected('alias_list_required')
                allowed = {'str', 'bool', 'int', 'float', 'timestamp'}
                aliases = []
                for value in value_node.value:
                    if not isinstance(value, yaml.ScalarNode) or value.tag.removeprefix('tag:yaml.org,2002:') not in allowed:
                        raise MetadataRejected('alias_text_required')
                    if not value.value.strip():
                        raise MetadataRejected('alias_empty')
                    aliases.append(value.value)
                # This field is text, regardless of YAML 1.1 implicit scalar typing.
                result[key] = aliases
            else:
                result[key] = self.construct_object(value_node, deep=deep)
        return result

def parse_frontmatter(note):
    if len(note.encode('utf-8')) > 16384:
        raise MetadataRejected('input_limit')
    lines = note.splitlines()
    fm, body_start = _split_frontmatter(lines)
    if lines and lines[0].strip() == '---' and body_start == 0:
        raise MetadataRejected('frontmatter_boundary')
    text = '\n'.join(fm) + ('\n' if fm else '')
    depth = events = 0
    try:
        for event in yaml.parse(text, Loader=yaml.SafeLoader):
            events += 1
            if isinstance(event, (yaml.MappingStartEvent, yaml.SequenceStartEvent)):
                depth += 1
            if isinstance(event, (yaml.MappingEndEvent, yaml.SequenceEndEvent)):
                depth -= 1
            if depth > 32 or events > 4096:
                raise MetadataRejected('parse_limit')
        loader = ContractLoader(text)
        try:
            node = loader.get_single_node()
            loader.root_node = node
            if node is None:
                data = {}
            elif not isinstance(node, yaml.MappingNode):
                raise MetadataRejected('mapping_required')
            else:
                data = loader.construct_document(node)
        finally:
            loader.dispose()
    except yaml.YAMLError as error:
        raise MetadataRejected('invalid_yaml') from error
    work = 0
    active = set()
    def normalize(value, depth=0):
        nonlocal work
        work += 1
        if work > 4096 or depth > 32:
            raise MetadataRejected('expansion_limit')
        if isinstance(value, (dict, list)):
            identity = id(value)
            if identity in active:
                raise MetadataRejected('cycle')
            active.add(identity)
            try:
                if isinstance(value, dict):
                    return {key: normalize(item, depth+1) for key, item in value.items()}
                return [normalize(item, depth+1) for item in value]
            finally:
                active.remove(identity)
        if isinstance(value, (datetime.datetime, datetime.date)):
            return value.isoformat()
        if value is None or isinstance(value, (str, bool, int)):
            return value
        if isinstance(value, float) and math.isfinite(value):
            return value
        raise MetadataRejected('unsupported_value')
    canonical = normalize(data)
    if len(json.dumps(canonical, ensure_ascii=False, allow_nan=False).encode('utf-8')) > 16384:
        raise MetadataRejected('output_limit')
    return canonical
